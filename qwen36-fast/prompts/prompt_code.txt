# ngram_cache.py - a simple in-memory n-gram cache with disk persistence.
# Used for prompt-lookup speculative decoding.

import json
import os
from collections import Counter, defaultdict
from typing import Dict, Iterable, List, Optional, Sequence, Tuple

DEFAULT_NGRAM_MIN = 1
DEFAULT_NGRAM_MAX = 4
DEFAULT_CACHE_PATH = os.path.expanduser("~/.cache/llama/ngram.bin")


class NgramCache:
    """A simple in-memory n-gram cache.

    The cache maps a tuple of token ids (the n-gram key) to a Counter of
    candidate next tokens. On lookup, it returns the most common next token
    for the longest matching n-gram. If no match is found at ngram_max, it
    backs off to ngram_max - 1 and so on down to ngram_min. The cache can be
    persisted to disk via save() and loaded via load().

    Parameters
    ----------
    ngram_min : int
        The minimum n-gram length to track.
    ngram_max : int
        The maximum n-gram length to track.
    """

    def __init__(self, ngram_min: int = DEFAULT_NGRAM_MIN, ngram_max: int = DEFAULT_NGRAM_MAX):
        if ngram_min < 1:
            raise ValueError("ngram_min must be >= 1")
        if ngram_max < ngram_min:
            raise ValueError("ngram_max must be >= ngram_min")
        self.ngram_min = ngram_min
        self.ngram_max = ngram_max
        self._data: Dict[Tuple[int, ...], Counter] = defaultdict(Counter)

    def update(self, tokens: Sequence[int]) -> None:
        """Update the cache with all n-grams from the given token sequence."""
        tokens = list(tokens)
        for n in range(self.ngram_min, self.ngram_max + 1):
            for i in range(len(tokens) - n):
                key = tuple(tokens[i:i + n])
                nxt = tokens[i + n]
                self._data[key][nxt] += 1

    def lookup(self, context: Sequence[int]) -> Optional[int]:
        """Return the most likely next token given the trailing context, or None."""
        context = list(context)
        for n in range(self.ngram_max, self.ngram_min - 1, -1):
            if len(context) < n:
                continue
            key = tuple(context[-n:])
            if key in self._data:
                most_common = self._data[key].most_common(1)
                if most_common:
                    return most_common[0][0]
        return None

    def save(self, path: str = DEFAULT_CACHE_PATH) -> None:
        """Persist the cache to disk as a JSON file."""
        os.makedirs(os.path.dirname(path), exist_ok=True)
        serialisable = {
            "ngram_min": self.ngram_min,
            "ngram_max": self.ngram_max,
            "data": [
                {"key": list(k), "counts": dict(v)}
                for k, v in self._data.items()
            ],
        }
        with open(path, "w") as f:
            json.dump(serialisable, f)

    @classmethod
    def load(cls, path: str = DEFAULT_CACHE_PATH) -> "NgramCache":
        """Load the cache from a JSON file on disk."""
        with open(path, "r") as f:
            data = json.load(f)
        cache = cls(ngram_min=data["ngram_min"], ngram_max=data["ngram_max"])
        for entry in data["data"]:
            cache._data[tuple(entry["key"])] = Counter({int(k): int(v) for k, v in entry["counts"].items()})
        return cache


def merge(a: NgramCache, b: NgramCache) -> NgramCache:
    """Merge two caches into a new one. Both must have the same ngram_min/max."""
    if a.ngram_min != b.ngram_min or a.ngram_max != b.ngram_max:
        raise ValueError("ngram_min/max must match to merge")
    out = NgramCache(ngram_min=a.ngram_min, ngram_max=a.ngram_max)
    for key, counts in a._data.items():
        out._data[key].update(counts)
    for key, counts in b._data.items():
        out._data[key].update(counts)
    return out
